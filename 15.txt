import { useState } from "react";
import "./App.css";
function App() {
const furniture = [
{
id: 1,
name: "Wooden Sofa",
category: "Living Room",
condition: "Refurbished - Excellent",
costPrice: 4500,
sellingPrice: 6500,
image: "https://via.placeholder.com/250x160?text=Wooden+Sofa"
},
{
id: 2,
name: "Office Chair",
category: "Office",
condition: "Refurbished - Good",
costPrice: 1200,
sellingPrice: 2200,
image: "https://via.placeholder.com/250x160?text=Office+Chair"
},
{
id: 3,
name: "Dining Table",
category: "Dining Room",
condition: "Refurbished - Excellent",
costPrice: 5000,
sellingPrice: 7800,
image: "https://via.placeholder.com/250x160?text=Dining+Table"
},
{
id: 4,
name: "Study Table",
category: "Study Room",
condition: "Refurbished - Good",
costPrice: 1800,
sellingPrice: 3000,
image: "https://via.placeholder.com/250x160?text=Study+Table"
},
{
id: 5,
name: "Queen Size Cot",
category: "Bedroom",
condition: "Refurbished - Excellent",
costPrice: 6000,
sellingPrice: 9000,
image: "https://via.placeholder.com/250x160?text=Queen+Size+Cot"
},
{
id: 6,
name: "Bookshelf",
category: "Storage",
condition: "Refurbished - Good",
costPrice: 1500,
sellingPrice: 2600,
image: "https://via.placeholder.com/250x160?text=Bookshelf"
}
];
const [cart, setCart] = useState([]);
const addToCart = (product) => {
const alreadyAdded = cart.find(
item => item.id === product.id
);
if (alreadyAdded) {
alert("This furniture is already in your cart.");
return;
}
setCart([...cart, product]);
};
const removeFromCart = (id) => {
const updatedCart = cart.filter(
item => item.id !== id
);
setCart(updatedCart);
};
const total = cart.reduce(
(sum, item) => sum + item.sellingPrice,
0
);
const placeOrder = () => {
if (cart.length === 0) {
alert("Your cart is empty.");
} else {
alert(
"Order Placed Successfully! Total Amount: ₹" + total
);
setCart([]);
}
};
return (
<div>
<header className="header">
<h1>ReNew Furniture</h1>
<p>
Quality Refurbished Furniture at Affordable Prices
</p>
</header>
<section className="about">
<h2>Give Furniture a Second Life</h2>
<p>
We acquire quality used furniture, refurbish it,
and make it available to customers at affordable prices.
</p>
</section>
<h2 className="section-title">
Furniture Available for Sale
</h2>
<div className="products">
{furniture.map(product => (
<div
className="product-card"
key={product.id}
>
<img
src={product.image}
alt={product.name}
/>
<h3>{product.name}</h3>
<p>
Category: {product.category}
</p>
<p>
Condition: {product.condition}
</p>
<h3>
₹{product.sellingPrice}
</h3>
<button
onClick={() => addToCart(product)}
>
Add to Cart
</button>
</div>
))}
</div>
<section className="cart">
<h2>Shopping Cart</h2>
{cart.length === 0 ? (
<p>No furniture added to the cart.</p>
) : (
cart.map(item => (
<div
className="cart-item"
key={item.id}
>
<span>
{item.name}
</span>
<span>
₹{item.sellingPrice}
</span>
<button
onClick={() =>
removeFromCart(item.id)
}
>
Remove
</button>
</div>
))
)}
<h2>
Total Amount: ₹{total}
</h2>
<button
className="order-button"
onClick={placeOrder}
>
Place Order
</button>
</section>
<footer>
<p>
© 2026 ReNew Furniture |
SustainableFurniture Marketplace
</p>
</footer>
</div>
);
}
export default App;
