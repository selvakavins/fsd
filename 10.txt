<script>
$(document).ready(function () {

    $(".star").click(function () {

        let rating = $(this).data("value");

        $(".star").removeClass("selected");

        $(".star").each(function () {

            if ($(this).data("value") <= rating) {
                $(this).addClass("selected");
            }

        });

        $("#result").text("You selected " + rating + " star(s)");
    });

});
